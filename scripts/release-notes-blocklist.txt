# Release Notes 敏感词黑名单
#
# 用途：scripts/release-win.py 在生成 release notes 后、发布前逐行扫描全文。
#       命中任一条即中止发布（退出码 1），必须人工确认并清理后才可继续。
#
# 规则：
#   - 一行一条；行首 `#` 为注释；空行忽略
#   - 大小写不敏感
#   - 每行按 **正则** 解释（子串匹配用普通文本即可）
#   - **本文件会随仓库公开分发**，因此黑名单本身不得写入完整真名/完整主机名，
#     只写「够识别的唯一片段」即可（姓名留姓+首字，主机名留前缀）
#
# 维护：发现新的敏感项追加到对应分组，提交信息里说明来源。

# ── 真实人名（只写可识别片段，不写全名）──────────────────────────
# 替换尖括号占位为你自己的片段；留空则该项不生效
<REAL_NAME_FRAGMENT_1>
<REAL_NAME_FRAGMENT_2>

# ── 联系方式 ─────────────────────────────────────────────────────
# QQ 号：以排除版本/日期/端口的方式命中「独立长数字」
#   - 前置要求非数字且非 v- 开头（避开 v5.1.5 里的 5.1.5 已被小数点隔断，无风险）
#   - 排除 4 位及以下（年份 2026 也需人工判断，故要求 6 位以上更安全）
#   - 行内不得含小数点（避开 5.1.5 / 1.2.3.4）
(?<![\d.])(?<![vV])[1-9]\d{5,11}(?![\d.])
# 中国大陆手机号
(?<!\d)1[3-9]\d{9}(?!\d)
# 微信号前缀
(?i)\bwxid_[0-9a-z]+

# ── 内网地址 / 主机名 ────────────────────────────────────────────
# 内网 IP（10.x / 172.16-31.x / 192.168.x）
\b10\.\d{1,3}\.\d{1,3}\.\d{1,3}\b
\b192\.168\.\d{1,3}\.\d{1,3}\b
\b172\.(?:1[6-9]|2\d|3[01])\.\d{1,3}\.\d{1,3}\b
# 本机开发机可识别片段
(?i)\by7000p\b
(?i)\b(?:asus|tuf)\s*gaming\b
# Windows 用户目录真实用户名（会带出本机账户名）
(?i)[a-z]:\\users\\(?!public\b|%username%|<)[^\\\s]+

# ── 云服 / 内部服务 ──────────────────────────────────────────────
# 替换为你的云服主机名片段
<CLOUD_HOST_FRAGMENT>
# 疑似凭证赋值
(?i)\b(?:api[_-]?key|secret|token|password)\s*[:=]\s*\S+

# ── 内部路径 / 内部分支名 ────────────────────────────────────────
# 绝对开发机路径
(?i)\b[a-z]:\\(?:my git|workbuddy|users)\\
# 内部工单分支名
(?i)\bworkorders-\d{4}-\d{2}-\d{2}(?:-[a-z0-9]+)?\b
